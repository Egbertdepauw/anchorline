# Dive Bailout Carry Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the broken global `cylinders.gasMix` model with a proper per-dive bailout-carry table so the bailout planner consumes the gases the diver actually carried on each dive.

**Architecture:** Drop `cylinders.gas_mix` (column + UI). Add `dive_bailout_carry` table keyed on `(dive_id, cylinder_id)` carrying gas, fill pressure, and ordering. Surface it as a list on Dive Detail → Equipment tab. The bailout planner reads carry rows for the current dive and produces `userBailoutTanks` + `userTankVolumesL` from them; empty list still falls through to Scenario B.

**Tech Stack:** Flutter, Drift (SQLite), PowerSync, Supabase Postgres. Existing test patterns: `test/data/db/equipment_dao_test.dart`, `test/analysis/bailout/bailout_planner_test.dart`.

/// Per-dive list of OC bailout cylinders the diver carried.
/// One row per (diveId, cylinderId); composite is encoded into [id]
/// as `<diveId>:<cylinderId>` so PowerSync can sync it under a single
/// primary key column.
class DiveBailoutCarry extends Table {
  TextColumn get id => text()();
  TextColumn get diveId => text().references(Dives, #id)();
  TextColumn get cylinderId => text().references(Cylinders, #id)();
  TextColumn get gasMix => text()(); // "21/35", "100", "32"
  IntColumn get fillPressureBar => integer().nullable()(); // null → use cylinder.workingPressureBar
  IntColumn get bailoutOrder => integer()();

  IntColumn get createdAt => integer()();
  IntColumn get updatedAt => integer()();
  IntColumn get deletedAt => integer().nullable()();
  IntColumn get version => integer().withDefault(const Constant(1))();
  IntColumn get syncStatus => integer().withDefault(const Constant(0))();
  TextColumn get ownerId => text().nullable()();

  @override
  Set<Column> get primaryKey => {id};
}
```

// ── DiveBailoutCarry ────────────────────────────────────────────────

Stream<void> watchDiveBailoutCarry(String diveId) =>
    _ps.watch(
      'SELECT 1 FROM dive_bailout_carry WHERE dive_id = ? AND deleted_at IS NULL LIMIT 1',
      parameters: [diveId],
    ).map((_) {});

Future<List<DiveBailoutCarryData>> listBailoutCarry(String diveId) async {
  final rows = await _ps.getAll(
    'SELECT * FROM dive_bailout_carry '
    'WHERE dive_id = ? AND deleted_at IS NULL '
    'ORDER BY bailout_order ASC',
    [diveId],
  );
  return rows.map(_rowToBailoutCarry).toList();
}

Future<List<({DiveBailoutCarryData carry, Cylinder cylinder})>>
    listBailoutCarryWithCylinders(String diveId) async {
  final rows = await _ps.getAll(
    'SELECT bc.*, c.* FROM dive_bailout_carry bc '
    'JOIN cylinders c ON c.id = bc.cylinder_id '
    'WHERE bc.dive_id = ? AND bc.deleted_at IS NULL '
    'ORDER BY bc.bailout_order ASC',
    [diveId],
  );
  return rows.map((r) => (
        carry: _rowToBailoutCarry(r),
        cylinder: _rowToCylinder(r),
      )).toList();
}

Future<String> addBailoutCarry({
  required String diveId,
  required String cylinderId,
  required String gasMix,
  int? fillPressureBar,
  required int bailoutOrder,
}) async {
  final id = '$diveId:$cylinderId';
  final now = DateTime.now().millisecondsSinceEpoch;
  await _ps.execute(
    'INSERT INTO dive_bailout_carry '
    '(id, dive_id, cylinder_id, gas_mix, fill_pressure_bar, bailout_order, '
    ' created_at, updated_at, deleted_at, version, sync_status) '
    'VALUES (?, ?, ?, ?, ?, ?, ?, ?, NULL, 1, 0)',
    [id, diveId, cylinderId, gasMix, fillPressureBar, bailoutOrder, now, now],
  );
  return id;
}

Future<void> updateBailoutCarry(
  String id, {
  String? gasMix,
  int? fillPressureBar,
  int? bailoutOrder,
}) async {
  final sets = <String>[];
  final args = <Object?>[];
  if (gasMix != null) { sets.add('gas_mix = ?'); args.add(gasMix); }
  if (fillPressureBar != null) { sets.add('fill_pressure_bar = ?'); args.add(fillPressureBar); }
  if (bailoutOrder != null) { sets.add('bailout_order = ?'); args.add(bailoutOrder); }
  if (sets.isEmpty) return;
  sets.add('updated_at = ?'); args.add(DateTime.now().millisecondsSinceEpoch);
  args.add(id);
  await _ps.execute(
    'UPDATE dive_bailout_carry SET ${sets.join(", ")} WHERE id = ?',
    args,
  );
}

Future<void> deleteBailoutCarry(String id) async {
  await _ps.execute(
    'UPDATE dive_bailout_carry SET deleted_at = ?, updated_at = ? WHERE id = ?',
    [DateTime.now().millisecondsSinceEpoch, DateTime.now().millisecondsSinceEpoch, id],
  );
}

DiveBailoutCarryData _rowToBailoutCarry(Map<String, dynamic> row) =>
    DiveBailoutCarryData(
      id: row['id'] as String,
      diveId: row['dive_id'] as String,
      cylinderId: row['cylinder_id'] as String,
      gasMix: row['gas_mix'] as String,
      fillPressureBar: row['fill_pressure_bar'] as int?,
      bailoutOrder: row['bailout_order'] as int,
      createdAt: row['created_at'] as int,
      updatedAt: row['updated_at'] as int,
      deletedAt: row['deleted_at'] as int?,
      version: row['version'] as int,
      syncStatus: row['sync_status'] as int,
      ownerId: row['owner_id'] as String?,
    );
```

Note: `DiveBailoutCarryData` is the Drift-generated row class name (Drift suffixes with `Data` for table classes when there's a collision risk).

```dart
DiveGasMix? _parseGasMix(String raw) {
  if (raw.isEmpty) return null;
  final parts = raw.split('/');
  final o2 = int.tryParse(parts[0]);
  if (o2 == null || o2 <= 0 || o2 > 100) return null;
  final he = parts.length > 1 ? (int.tryParse(parts[1]) ?? 0) : 0;
  if (he < 0 || o2 + he > 100) return null;
  return DiveGasMix(oxygenPercent: o2, heliumPercent: he, isDiluent: false);
}
```

// lib/ui/widgets/dive_bailout_carry_section.dart
class DiveBailoutCarrySection extends StatelessWidget {
  final String diveId;
  final EquipmentDao dao;

  const DiveBailoutCarrySection({
    super.key,
    required this.diveId,
    required this.dao,
  });

  @override
  Widget build(BuildContext context) {
    return StreamBuilder<void>(
      stream: dao.watchDiveBailoutCarry(diveId),
      builder: (context, _) => FutureBuilder<List<({DiveBailoutCarryData carry, Cylinder cylinder})>>(
        future: dao.listBailoutCarryWithCylinders(diveId),
        builder: (context, snap) {
          final rows = snap.data ?? const [];
          return Column(
            crossAxisAlignment: CrossAxisAlignment.stretch,
            children: [
              const SectionHeader('Bailout carry'),
              if (rows.isEmpty)
                const Padding(
                  padding: EdgeInsets.symmetric(vertical: 12),
                  child: Text('No bailout cylinders configured'),
                ),
              for (final r in rows)
                _CarryRow(
                  carry: r.carry,
                  cylinder: r.cylinder,
                  onEditGas: (newGas) => dao.updateBailoutCarry(r.carry.id, gasMix: newGas),
                  onEditFill: (newFill) => dao.updateBailoutCarry(r.carry.id, fillPressureBar: newFill),
                  onDelete: () => dao.deleteBailoutCarry(r.carry.id),
                ),
              TextButton.icon(
                icon: const Icon(Icons.add),
                label: const Text('Add cylinder'),
                onPressed: () async {
                  final picked = await _showCylinderPicker(context, dao);
                  if (picked == null) return;
                  await dao.addBailoutCarry(
                    diveId: diveId,
                    cylinderId: picked.id,
                    gasMix: '21', // default; user can edit inline immediately
                    bailoutOrder: rows.length + 1,
                  );
                },
              ),
            ],
          );
        },
      ),
    );
  }
}

// `_CarryRow` shows: bailoutOrder · cylinder.label · editable gasMix · editable fillBar · delete X
// `_showCylinderPicker` returns the chosen Cylinder from a modal bottom sheet listing dao.listCylinders().
